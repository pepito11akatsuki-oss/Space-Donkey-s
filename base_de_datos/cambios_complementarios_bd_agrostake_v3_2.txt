=====================================================================
CAMBIOS COMPLEMENTARIOS EN BASE DE DATOS PARA AGROSTAKE (v3.2)
=====================================================================

Estos pasos se ejecutan sobre una base nueva después de cargar el archivo
principal `agrostake_schema_v3_2.sql`. No alteran tablas, columnas ni
restricciones existentes.

---------------------------------------------------------------------
PASO 1: CREACIÓN DE ROLES Y BASE DE DATOS (Ejecutar como superusuario)
---------------------------------------------------------------------
-- El rol de aplicación NO debe ser dueño de las tablas ni superusuario
-- para que PostgreSQL aplique estrictamente Row-Level Security (RLS).

CREATE ROLE agrostake_owner WITH LOGIN PASSWORD 'owner_pass_local';
CREATE ROLE agrostake_app   WITH LOGIN PASSWORD 'app_pass_local' NOSUPERUSER NOCREATEDB NOCREATEROLE NOBYPASSRLS;
CREATE DATABASE agrostake OWNER agrostake_owner;

-- (Aquí debes cargar agrostake_schema_v3_2.sql usando agrostake_owner)
-- psql -U agrostake_owner -d agrostake -v ON_ERROR_STOP=1 -f agrostake_schema_v3_2.sql

-- 2. Función SECURITY DEFINER para validar refresh_token antes de fijar app.org_id
CREATE OR REPLACE FUNCTION auth_lookup_refresh_token(p_token_hash TEXT)
RETURNS TABLE (
    token_id   UUID,
    user_id    UUID,
    org_id     UUID,
    role       user_role,
    expires_at TIMESTAMPTZ,
    revoked_at TIMESTAMPTZ
)
LANGUAGE sql SECURITY DEFINER SET search_path = public AS $$
    SELECT rt.id, u.id, u.org_id, u.role, rt.expires_at, rt.revoked_at
    FROM refresh_tokens rt
    JOIN users u ON u.id = rt.user_id
    WHERE rt.token_hash = p_token_hash
    LIMIT 1;
$$;

-- 3. Función SECURITY DEFINER para resolver el org_id de una estaca IoT por device_uid
CREATE OR REPLACE FUNCTION iot_resolve_stake_context(p_device_uid TEXT)
RETURNS TABLE (
    stake_id  UUID,
    org_id    UUID,
    parcel_id UUID,
    is_active BOOLEAN
)
LANGUAGE sql SECURITY DEFINER SET search_path = public AS $$
    SELECT id, org_id, parcel_id, is_active
    FROM stakes
    WHERE device_uid = trim(p_device_uid)
    LIMIT 1;
$$;

PASO 3: PERMISOS MÍNIMOS PARA EL BACKEND (Ejecutar como agrostake_owner)
---------------------------------------------------------------------

GRANT CONNECT ON DATABASE agrostake TO agrostake_app;
GRANT USAGE ON SCHEMA public TO agrostake_app;
GRANT SELECT, INSERT, UPDATE, DELETE ON ALL TABLES IN SCHEMA public TO agrostake_app;
GRANT USAGE, SELECT ON ALL SEQUENCES IN SCHEMA public TO agrostake_app;
GRANT EXECUTE ON ALL FUNCTIONS IN SCHEMA public TO agrostake_app;
